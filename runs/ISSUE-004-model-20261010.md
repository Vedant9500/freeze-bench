import math, sys
sys.path.insert(0, "/tmp/eval/ISSUE-004")
from openlmlib.write_gate import WriteGate
from openlmlib.schema import ValidationIssue

class StubEmbedder:
    def __init__(self, claim, sim):
        self.claim = claim; self.sim = sim
    def encode(self, texts):
        out = []
        for t in texts:
            if t == self.claim:
                out.append([1.0, 0.0])
            else:
                s = self.sim
                out.append([s, math.sqrt(max(0.0, 1 - s*s))])
        return out

REASON = "This reasoning is long enough to pass the twenty char minimum gate for sure, explaining support."
def check(name, claim, ev, sim, threshold=0.6, trusted=False):
    if trusted:
        g = WriteGate(0.6, 20, 0.0, 0.85, 5, embedder=None, vector_store=None, finding_lookup=None)
        issues = g.validate(claim, ev, REASON, 0.9)
        issues.append(ValidationIssue(field="export", message="trusted_export: claim/evidence similarity checks skipped", severity="warning"))
        adj = 0.9  # library.py:660-661
    else:
        g = WriteGate(0.6, 20, threshold, 0.85, 5, embedder=StubEmbedder(claim, sim))
        issues = g.validate(claim, ev, REASON, 0.9)
        adj = g.adjust_confidence(claim, ev, 0.9, issues)
        sim = g.evidence_similarity(claim, ev)
    status = "rejected" if not WriteGate.is_allowed(issues) else ("ok-clean" if not any(i.field=="evidence" and i.severity=="warning" for i in issues) else "ok-with-warning(evidence)")
    if trusted: status = "ok-with-warning(export+embedding)"
    print(f"{name}: sim={sim:.2f} -> {status} 0.900->{adj:.3f} delta={adj-0.9:+.3f} {[(i.field,i.severity) for i in issues]}")

C = "Redis caching reduces API latency under production load"
check("verbatim", C, [C], 1.0)
check("paraphrase", C, ["Enabling the Redis cache lowers production API response times"], 0.72)
check("unrelated", C, ["Penguins migrate south in winter across the ice shelf"], 0.12)
check("smuggling-trusted", "Quantum flux cures baldness in three days", ["Penguins migrate south in winter across the ice shelf"], 0.05, trusted=True)
